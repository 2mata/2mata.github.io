{{- /* Render hook gambar — pemetaan komponen: 1. Gambar eksternal (http/https) -> (apa adanya) 2. Gambar lokal raster di assets/ -> : source foo.webp (pre-generate q80 1:1) + fallback file asli 3. Gambar lokal non-raster (svg/gif) -> biasa (GIF tidak dikonversi agar animasi tetap utuh) 4. Gambar masih di static/ (belum pindah) -> biasa + width/height dari imageConfig Catatan: lebar/tinggi selalu ditulis agar tidak terjadi layout shift (CLS). */ -}} {{- $src := .Destination | safeURL -}} {{- $rel := strings.TrimPrefix "/" .Destination -}} {{- /* alt diambil dari caption markdown, dibersihkan dari karakter non-huruf */ -}} {{- $caption := .Text | lower -}} {{- $replacere := replaceRE "[^a-zA-Z\\s]" "" $caption -}} {{- $trim := trim $replacere " " -}} {{- $split := split $trim " " | first 3 -}} {{- $alt := delimit $split " " -}} {{- if strings.HasPrefix .Destination "http" -}} {{- else -}} {{- $res := resources.Get $rel -}} {{- /* WebP q80 1:1 sudah digenerate sekali di luar build dan disimpan bersebelahan dengan file aslinya: assets/…/foo.png -> assets/…/foo.webp. Jadi build tidak lagi menjalankan .Process "webp" (di mesin ini 14 detik/gambar). */ -}} {{- $webp := resources.Get (printf "%s.webp" (strings.TrimSuffix (path.Ext $rel) $rel)) -}} {{- if and $res $webp -}} {{- else if $res -}} {{- else -}} {{- /* Fallback: gambar masih di static/. Guard os.FileExists wajib karena imageConfig MENGGAGALKAN seluruh build bila file tidak ditemukan. */ -}} {{- $config := dict "Width" 0 "Height" 0 -}} {{- if os.FileExists (printf "static/%s" $rel) -}} {{- $config = imageConfig (printf "/static/%s" $rel) -}} {{- end -}} {{- end -}} {{- end -}}
