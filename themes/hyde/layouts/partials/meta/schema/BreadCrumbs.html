{{/* Breadcrumb JSON-LD — dipakai di single.html. Nama tiap item diambil dari Title halaman aslinya (mis. "Blog", "Cara Sadap WhatsApp"), bukan slug URL-nya. Google menampilkan nama ini apa adanya di hasil pencarian, jadi slug seperti "aplikasi-sadap-mspy" kurang informatif. Slug hanya dipakai sebagai fallback. */}} {{/* Item pertama: beranda */}} {{ $items := slice (dict "url" site.BaseURL "name" "Beranda") }} {{/* Buang baseURL dan garis miring di ujung, lalu pecah per segmen: "blog/cara-sadap-wa/" -> ["blog", "cara-sadap-wa"] */}} {{ $slug := replace .Permalink site.BaseURL "" | strings.TrimSuffix "/" }} {{ $path := "" }} {{ range $element := split $slug "/" }} {{ $path = printf "%s/%s" $path $element }} {{/* Coba cocokkan segmen dengan halaman aslinya agar dapat Title + URL kanonik */}} {{ $name := "" }} {{ $url := "" }} {{ with site.GetPage $path }} {{ $name = .Title }} {{ $url = .Permalink }} {{ end }} {{/* Fallback: slug diubah jadi kata biasa, mis. "cara-sadap-wa" -> "Cara Sadap Wa" */}} {{ if not $name }}{{ $name = $element | replaceRE `-` " " | title }}{{ end }} {{ if not $url }}{{ $url = absURL (strings.TrimPrefix "/" $path) }}{{ end }} {{ $items = $items | append (dict "url" $url "name" $name) }} {{ end }}
